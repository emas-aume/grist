function formatValue(value, columnId = "") { if ( value === null || value === undefined || value === "" ) { return ""; } /* ========================== NUMÉRO FE ========================== Fait Nº 2026-011-5574411 devient 5574411 */ if (columnId === "Numero_FE") { const texte = String(value); const match = texte.match(/(\d+)\s*$/); if (match) { return match[1]; } return texte; } /* ========================== OBJET DATE JAVASCRIPT ========================== */ if (value instanceof Date) { const jour = String(value.getUTCDate()).padStart(2, "0"); const mois = String(value.getUTCMonth() + 1).padStart(2, "0"); const annee = String(value.getUTCFullYear()).slice(-2); return `${jour}/${mois}/${annee}`; } /* ========================== DATES ISO ========================== Fonctionne avec : 2026-08-31T00:00:00.000Z mais aussi : "2026-08-31T00:00:00.000Z" */ let texte = String(value).trim(); /* Supprime d'éventuels guillemets autour de la valeur */ texte = texte.replace( /^["']|["']$/g, "" ); const dateMatch = texte.match( /^(\d{4})-(\d{2})-(\d{2})(?:T.*)?$/ ); if (dateMatch) { const annee = dateMatch[1].slice(-2); const mois = dateMatch[2]; const jour = dateMatch[3]; return `${jour}/${mois}/${annee}`; } /* ========================== LISTES GRIST ========================== */ if (Array.isArray(value)) { if (value[0] === "L") { return value .slice(1) .join(", "); } return value.join(", "); } /* ========================== AUTRES OBJETS ========================== */ if ( typeof value === "object" ) { return JSON.stringify(value); } /* ========================== VALEUR NORMALE ========================== */ return texte; }
